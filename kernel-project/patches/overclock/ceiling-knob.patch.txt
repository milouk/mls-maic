    maic: cpufreq: hard ceiling knob so the OC OPPs stay dormant until raised deliberately
    
    The overclock patch clamps policy->max only in ->init(), so the default ceiling
    was 1300 MHz at boot but not afterwards: any later writer of scaling_max_freq --
    MTK PerfService restoring the hardware max when a touch/launch boost ends, the
    thermal manager un-throttling -- lands on cpuinfo_max_freq, now 1500000. Seen
    on the device on 2026-09-17: 69 s at 1500 MHz @ 1.35 V under the performance
    governor with nobody having asked for it (time_in_state), before the 1400 MHz
    soak pinned the frequency.
    
    Add maic_oc_ceiling_khz (module parameter, default 1300000, sysfs
    /sys/module/mt8167_cpufreq/parameters/) and enforce it in a ->verify()
    wrapper: every policy change (sysfs, thermal, perf hints) passes through
    verify, so the ceiling holds until the knob itself is raised -- which only the
    soak/adoption scripts do, on purpose. min is folded under max if needed.

+#ifdef CONFIG_MAIC_CPU_OC
+/*
+ * MAIC: hard ceiling for policy->max. The ->init() clamp above only sets the
+ * default; anything that later writes scaling_max_freq (MTK PerfService after a
+ * boost, thermal un-throttle) would land on cpuinfo_max_freq = 1500000. Every
+ * policy change passes through ->verify(), so enforce the ceiling here. Raise
+ * deliberately: echo 1500000 > /sys/module/mt8167_cpufreq/parameters/maic_oc_ceiling_khz
+ */
+static unsigned int maic_oc_ceiling_khz = 1300000;
+module_param(maic_oc_ceiling_khz, uint, 0644);
+MODULE_PARM_DESC(maic_oc_ceiling_khz, "MAIC: hard cpufreq ceiling in kHz (default 1300000)");
+
+static int maic_cpufreq_verify(struct cpufreq_policy *policy)
+{
+	if (maic_oc_ceiling_khz && policy->max > maic_oc_ceiling_khz)
+		policy->max = maic_oc_ceiling_khz;
+	if (policy->min > policy->max)
+		policy->min = policy->max;
+	return cpufreq_generic_frequency_table_verify(policy);
+}
+#endif /* CONFIG_MAIC_CPU_OC */
